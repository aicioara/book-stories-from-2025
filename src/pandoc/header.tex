% Macro
\newcommand{\partblankleft}[1]{%
  \clearpage
  % If we're about to start an odd page now, it means the previous page was even
  % (so we'd otherwise see content facing the \part). Insert a blank odd page.
  \ifodd\value{page}%
    \thispagestyle{empty}\null\newpage
  \fi
  % Now insert the blank even (left) page that will face the \part page.
  \thispagestyle{empty}\null\newpage
  % Now we're on an odd (right) page: place the part.
  \part{#1}%
}
